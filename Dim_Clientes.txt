let
    // Paso 1: Conexión al archivo Excel
    Origen = Excel.Workbook(File.Contents("C:\Users\Usuario\Desktop\ANALISIS DE DATOS\Clase 6\Pipeline_ETL_Dataset.xlsx"), null, false),

    // Paso 2: Selección de la hoja clientes
    clientes_sheet = Origen{[Item = "clientes", Kind = "Sheet"]}[Data],

    // Paso 3: Filtrar contenido nulo y espacios en blanco
    // Eliminamos valores vacíos o nulos en la carga inicial
    FiltrarContenido = each List.Select(_, each _ <> null and (not (_ is text) or Text.Trim(_) <> "")),

    // Paso 4: Quitar filas inferiores vacías
    FilasInferioresQuitadas = Table.RemoveLastN(clientes_sheet, each try List.IsEmpty(List.Skip(FiltrarContenido(Record.FieldValues(_)), 1)) otherwise false),

    // Paso 5: Promover encabezados
    EncabezadosPromovidos = Table.PromoteHeaders(FilasInferioresQuitadas, [PromoteAllScalars=true]),

    // Paso 6: Tipar columnas iniciales
    TipadoInicial = Table.TransformColumnTypes(EncabezadosPromovidos,{
        {"id_cliente", Int64.Type}, 
        {"nombre_cliente", type text}, 
        {"email", type text}, 
        {"ciudad", type text}, 
        {"pais", type text}, 
        {"segmento", type text}, 
        {"fecha_registro", Int64.Type}
    }),

    // Paso 7: Eliminar duplicados por id_cliente
    // Justificación: el sistema origen duplica registros al reintentar el alta
    DuplicadosQuitados = Table.Distinct(TipadoInicial, {"id_cliente"}),

    // Paso 8: Eliminar filas completamente en blanco
    FilasEnBlancoEliminadas = Table.SelectRows(DuplicadosQuitados, each not List.IsEmpty(List.RemoveMatchingItems(Record.FieldValues(_), {"", null}))),

    // Paso 9: Filtrar filas con email nulo
    // Justificación: email es campo crítico, no se puede conservar sin valor
    FilasFiltradas = Table.SelectRows(FilasEnBlancoEliminadas, each ([email] <> null)),

    // Paso 10: Reemplazar ciudad nula por "Sin dato"
    // Justificación: mantener la fila pero marcar ausencia de información
    ValorReemplazado = Table.ReplaceValue(FilasFiltradas,null,"Sin dato",Replacer.ReplaceValue,{"ciudad"}),

    // Paso 11: Tipar fecha_registro como tipo date
    TipadoFinal = Table.TransformColumnTypes(ValorReemplazado,{
        {"fecha_registro", type date}, 
        {"id_cliente", Int64.Type}
    })
in
    TipadoFinal
