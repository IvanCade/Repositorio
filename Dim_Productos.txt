let
    // Paso 1: Conexión al archivo Excel
    Origen = Excel.Workbook(File.Contents("C:\Users\Usuario\Desktop\ANALISIS DE DATOS\Clase 6\Pipeline_ETL_Dataset.xlsx"), null, false),

    // Paso 2: Selección de la hoja productos
    productos_sheet = Origen{[Item = "productos", Kind = "Sheet"]}[Data],

    // Paso 3: Filtrar contenido nulo y espacios en blanco
    // Eliminamos valores vacíos o nulos en la carga inicial
    FiltrarContenido = each List.Select(_, each _ <> null and (not (_ is text) or Text.Trim(_) <> "")),

    // Paso 4: Quitar filas inferiores vacías
    FilasInferioresQuitadas = Table.RemoveLastN(productos_sheet, each try List.IsEmpty(List.Skip(FiltrarContenido(Record.FieldValues(_)), 1)) otherwise false),

    // Paso 5: Promover encabezados
    EncabezadosPromovidos = Table.PromoteHeaders(FilasInferioresQuitadas, [PromoteAllScalars=true]),

    // Paso 6: Tipar columnas iniciales
    TipadoInicial = Table.TransformColumnTypes(EncabezadosPromovidos,{
        {"id_producto", Int64.Type}, 
        {"nombre_producto", type text}, 
        {"categoria", type text}, 
        {"subcategoria", type text}, 
        {"precio", Int64.Type}, 
        {"costo", Int64.Type}, 
        {"stock", Int64.Type}, 
        {"activo", Int64.Type}
    }),

    // Paso 7: Eliminar duplicados por id_producto
    // Justificación: el sistema origen duplicó registros al cargar productos
    DuplicadosQuitados = Table.Distinct(TipadoInicial, {"id_producto"}),

    // Paso 8: Reemplazar nulos en precio por 0
    // Justificación: conservamos el producto para mantener integridad de las ventas,
    // aunque no tenga precio válido; se marca como 0 para que no aporte ingresos
    PrecioReemplazado = Table.ReplaceValue(DuplicadosQuitados,null,0,Replacer.ReplaceValue,{"precio"}),

    // Paso 9: Reemplazar nulos en categoria por "Sin categoría"
    // Justificación: mantener la fila pero marcar ausencia de información
    CategoriaReemplazada = Table.ReplaceValue(PrecioReemplazado,null,"Sin categoría",Replacer.ReplaceValue,{"categoria"}),

    // Paso 10: Tipar columnas finales
    TipadoFinal = Table.TransformColumnTypes(CategoriaReemplazada,{
        {"id_producto", Int64.Type}, 
        {"precio", type number}, 
        {"costo", type number}, 
        {"stock", Int64.Type}
    })
in
    TipadoFinal
